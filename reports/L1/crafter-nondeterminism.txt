import collections, hashlib, json, random, sys
import crafter
from crafter import engine

class OrderedSet(dict):
    """Insertion-ordered stand-in for the per-chunk object set: add/remove/iterate as set does."""
    def add(self, x): self[x] = None
    def remove(self, x): del self[x]

if sys.argv[1] == "ordered":
    _reset = engine.World.reset
    def reset(self, seed=None):
        _reset(self, seed)
        self._chunks = collections.defaultdict(OrderedSet)
    engine.World.reset = reset

seed, steps = int(sys.argv[2]), int(sys.argv[3])
env = crafter.Env(seed=seed); env.reset()
removed = []
_remove = env._world.remove
def remove(obj):
    removed.append(type(obj).__name__); _remove(obj)
env._world.remove = remove
rng = random.Random(seed)
for t in range(1, steps + 1):

    print(t, removed[n:] or "-", h)
    if done: break
