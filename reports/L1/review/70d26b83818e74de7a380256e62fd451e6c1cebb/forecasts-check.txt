import json, hashlib, math, sys, re
rows=[]
for p in sys.argv[1:]:
    rows += [json.loads(l) for l in open(p) if l.strip()]
claims={r["id"]:r for r in rows if "claim" in r}
bad=[i for i,r in claims.items() if hashlib.sha256((r["claim"]+r["ts"]).encode()).hexdigest()[:12]!=i]
print("claims",len(claims),"id mismatches",bad)
res=[r for r in rows if "outcome" in r and "id" in r]
print("resolution rows with id",len(res),"rows w/o id", sum(1 for r in rows if "outcome" in r and "id" not in r))
orphan=[r["id"] for r in res if r["id"] not in claims]; print("orphans",orphan)
seen={}; 
for r in res: seen.setdefault(r["id"],r)
def typ(c):
    c=c.lower()
    if "pins" in c and "green" in c: return "pins"
    if "reviewer verdict" in c or "accept" in c: return "review"
    if "host verdict" in c: return "host"
    return "other:"+c[:40]
ll=0; ll5=0; hist={}
for i,r in sorted(seen.items(), key=lambda kv: kv[1]["ts"]):
    c=claims[i]; y=bool(r["outcome"]); p=c["p"]; t=typ(c["claim"])
    h=hist.get(t,[]); b=(sum(h)/len(h)) if h else 0.5
    def L(q): 
        q=q if y else 1-q
        return -math.log(q) if q>0 else math.inf
    ll+=L(p); ll5+=math.log(2); 
    print(f"{i} {t:8} p={p} y={y} base={b:.3f} Lp={L(p):.3f} Lb={L(b):.3f}")
    hist.setdefault(t,[]).append(y)
    r["_lb"]=L(b)
print(f"resolved {len(seen)}; open {len(set(claims)-set(seen))}: {sorted(set(claims)-set(seen))}")
print(f"log-loss {ll:.3f}; vs 0.5 {ll5:.3f}; vs empirical-by-type {sum(r['_lb'] for r in seen.values()):.3f}")
