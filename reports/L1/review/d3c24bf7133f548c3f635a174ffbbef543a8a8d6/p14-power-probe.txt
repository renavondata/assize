    def rate(acts):
      hits=0
      for _ in range(T):
        c=[0]*k
        for _ in range(N): c[rng.choice(acts)]+=1
        e=N/k; x=sum((ci-e)**2/e for ci in c)
        hits+= _chi2_sf(x,k-1)<1e-6

    # Empirical false-refusal rate of P14's uniformity test under the true null (uniform over 17, 200 draws).
    import sys, numpy as np
    sys.path.insert(0, '.')
    from harness.pins import _chi2_sf
    lo, hi = 0.0, 200.0
    for _ in range(100):
        m = (lo + hi) / 2
        lo, hi = (m, hi) if _chi2_sf(m, 16) >= 1e-6 else (lo, m)
    crit = hi
    rng = np.random.default_rng(20261008)
    k, N, e = 17, 200, 200 / 17
    total, hits = 0, 0
    for _ in range(40):
        c = rng.multinomial(N, [1 / k] * k, size=500_000)
        x = ((c - e) ** 2 / e).sum(axis=1)
        hits += int((x > crit).sum()); total += len(x)
    print(f"critical chi2 (df 16, alpha 1e-6) = {crit:.3f}")
    print(f"null draws {total}; refused {hits}; empirical false-refusal rate {hits/total:.2e} vs nominal 1e-6")
