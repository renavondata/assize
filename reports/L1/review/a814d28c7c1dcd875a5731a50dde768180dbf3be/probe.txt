"""Reviewer probe (independent of the Builder's): trajectory hash for crafter 1.8.3, stock or with
insertion-ordered chunk containers. argv: mode seed steps reps. Prints one hash per rep (same process)."""
import collections, hashlib, random, sys
import crafter
from crafter import engine

class Ordered(dict):
    def add(self, x): self[x] = None
    def remove(self, x): del self[x]

mode, seed, steps, reps = sys.argv[1], int(sys.argv[2]), int(sys.argv[3]), int(sys.argv[4])
if mode == "ordered":
    _r = engine.World.reset
    def reset(self, seed=None):
        _r(self, seed); self._chunks = collections.defaultdict(Ordered)
    engine.World.reset = reset

def run():
    env = crafter.Env(seed=seed); env.reset(); acts = random.Random(1000 + seed); h = hashlib.sha256(); n = 0
    for n in range(1, steps + 1):
        _, r, done, info = env.step(acts.randrange(17))
        h.update(info["semantic"].tobytes()); h.update(repr(sorted(info["inventory"].items())).encode()); h.update(repr(r).encode())
        if done: break
    return f"{h.hexdigest()[:12]}@{n}"

print(mode, seed, *[run() for _ in range(reps)])
